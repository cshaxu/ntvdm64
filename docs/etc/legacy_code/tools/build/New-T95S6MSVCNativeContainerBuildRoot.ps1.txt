[CmdletBinding()]
param(
    [string]$RepositoryRoot = '',
    [string]$BuildRoot = '',
    [string]$ManifestPath = '',
    [string]$ProjectionConfig = ''
)

$ErrorActionPreference = 'Stop'

if ([string]::IsNullOrWhiteSpace($RepositoryRoot)) {
    $scriptPath = $MyInvocation.MyCommand.Path
    if ([string]::IsNullOrWhiteSpace($scriptPath)) { throw 'RepositoryRoot was not supplied and script path is unavailable.' }
    $RepositoryRoot = Split-Path -Parent (Split-Path -Parent $scriptPath)
}
$repository = [IO.Path]::GetFullPath($RepositoryRoot)
if ([string]::IsNullOrWhiteSpace($BuildRoot)) {
    $BuildRoot = Join-Path $repository 'artifacts\build\bochs-2.6-native-nogui-msvc-cpu3-r1'
}
if ([string]::IsNullOrWhiteSpace($ManifestPath)) {
    $ManifestPath = Join-Path $repository 'artifacts\analysis\t95-s6-msvc-native-container-manifest-001-20260811-001\native-container-manifest.json'
}
if ([string]::IsNullOrWhiteSpace($ProjectionConfig)) {
    $ProjectionConfig = Join-Path $repository 'artifacts\analysis\t95-s6-msvc-x86-config-projection-003-20260811-001\config.h'
}

$build = [IO.Path]::GetFullPath($BuildRoot)
$source = Join-Path $repository 'refs\bochs'
$manifestFile = [IO.Path]::GetFullPath($ManifestPath)
$projection = [IO.Path]::GetFullPath($ProjectionConfig)
if (Test-Path -LiteralPath $build) { throw "Refusing to overwrite existing build directory: $build" }
foreach ($input in @($source, $manifestFile, $projection)) {
    if (-not (Test-Path -LiteralPath $input)) { throw "Required build-shim input is missing: $input" }
}

$manifest = Get-Content -LiteralPath $manifestFile -Raw | ConvertFrom-Json
if ($manifest.schema -ne 'ntdos64.t95-s6.native-container-manifest.v1') { throw 'Unsupported native-container manifest schema.' }
function Assert-ManifestHash([string]$Path, [string]$Expected, [string]$Label) {
    $actual = (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToUpperInvariant()
    if ($actual -ne $Expected.ToUpperInvariant()) { throw "$Label hash mismatch: expected $Expected, got $actual." }
}
Assert-ManifestHash (Join-Path $source 'config.h') $manifest.inputs.importedConfig.sha256 'Pinned source config'
Assert-ManifestHash (Join-Path $source 'Makefile') $manifest.inputs.originalMakefile.sha256 'Original Makefile'
Assert-ManifestHash (Join-Path $source 'vs2008\bochs.vcproj') $manifest.inputs.originalExecutableProject.sha256 'Original executable project'
Assert-ManifestHash $projection $manifest.inputs.cpu3Projection.sha256 'CPU3 projection'

$sourceMakefile = Join-Path $source 'Makefile'
$makeLines = [IO.File]::ReadAllLines($sourceMakefile)
$targetStart = [Array]::FindIndex([string[]]$makeLines, [Predicate[string]]{ param($line) $line -match '^bochs\.exe:' })
if ($targetStart -lt 0) { throw 'Original Makefile does not contain the bochs.exe target.' }
$targetEnd = -1
for ($index = $targetStart + 1; $index -lt $makeLines.Count; $index++) {
    if ([string]::IsNullOrWhiteSpace($makeLines[$index])) { $targetEnd = $index - 1; break }
}
if ($targetEnd -lt $targetStart) { throw 'Original Makefile bochs.exe target has no terminating blank line.' }
$targetLines = @($makeLines[$targetStart..$targetEnd])
$targetLines[0] = $targetLines[0] -replace '^bochs\.exe:', 'ntdos64-native-container.exe:'
if (($targetLines -join "`n") -match '(?i)\bbochs\.exe\b') { throw 'Derived target retained a bochs.exe reference.' }
if (($targetLines -join "`n") -notmatch '(?m)^\s*link\s+') { throw 'Derived target did not retain the original link command.' }

$romSource = Join-Path $source 'bios'
$profileSource = Join-Path (Split-Path -Parent $manifestFile) 'native-container-profile.bochsrc'
foreach ($input in @((Join-Path $romSource 'BIOS-bochs-latest'), (Join-Path $romSource 'VGABIOS-lgpl-latest'), $profileSource)) {
    if (-not (Test-Path -LiteralPath $input -PathType Leaf)) { throw "Declared profile input is missing: $input" }
}

New-Item -ItemType Directory -Path $build -Force | Out-Null
& robocopy $source $build /E /COPY:DAT /DCOPY:DAT /R:1 /W:1 /XJ /NFL /NDL /NJH /NJS | Out-Null
if ($LASTEXITCODE -gt 7) { throw "robocopy failed with exit code $LASTEXITCODE" }

Copy-Item -LiteralPath $projection -Destination (Join-Path $build 'config.h') -Force
New-Item -ItemType Directory -Path (Join-Path $build 'ROM') -Force | Out-Null
Copy-Item -LiteralPath (Join-Path $romSource 'BIOS-bochs-latest') -Destination (Join-Path $build 'ROM\BIOS-bochs-latest') -Force
Copy-Item -LiteralPath (Join-Path $romSource 'VGABIOS-lgpl-latest') -Destination (Join-Path $build 'ROM\VGABIOS-lgpl-latest') -Force
Copy-Item -LiteralPath $profileSource -Destination (Join-Path $build 'native-container-profile.bochsrc') -Force

$shim = @(
    '# Generated by New-T95S6MSVCNativeContainerBuildRoot.ps1.',
    '# Use only: nmake /n /f ntdos64-native-container.mak ntdos64-native-container.exe',
    '# The original Makefile is included unchanged; this target is a renamed mechanical extraction of its bochs.exe recipe.',
    '!INCLUDE Makefile',
    ''
) + $targetLines + @('')
$shimPath = Join-Path $build 'ntdos64-native-container.mak'
[IO.File]::WriteAllText($shimPath, ($shim -join "`r`n"), [Text.UTF8Encoding]::new($false))

$record = [ordered]@{
    schema = 'ntdos64.t95-s6.msvc-native-container-build-root.v1'
    source = $source
    buildRoot = $build
    manifest = $manifestFile
    projection = $projection
    copiedRootConfigSha256 = (Get-FileHash -LiteralPath (Join-Path $build 'config.h') -Algorithm SHA256).Hash
    profileSha256 = (Get-FileHash -LiteralPath (Join-Path $build 'native-container-profile.bochsrc') -Algorithm SHA256).Hash
    shimSha256 = (Get-FileHash -LiteralPath $shimPath -Algorithm SHA256).Hash
    target = 'ntdos64-native-container.exe'
    permittedNextCommand = 'nmake /n /f ntdos64-native-container.mak ntdos64-native-container.exe'
    prohibited = @('all', 'bochs.exe', 'devenv', 'msbuild solution', 'configure', 'compile', 'link execution', 'adapter', 'cli', 'opennt')
}
[IO.File]::WriteAllText((Join-Path $build 'ntdos64-native-container-build-root.json'), ($record | ConvertTo-Json -Depth 5), [Text.UTF8Encoding]::new($false))
Write-Host "Prepared fresh MSVC native-container build root: $build"
