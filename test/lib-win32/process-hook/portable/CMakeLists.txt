cmake_minimum_required(VERSION 3.20)
# Keep compiler checks aligned with the release test configuration.
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(process_hook_portable_tests C)
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(P "${ROOT}/src/lib-win32/process-hook")
get_filename_component(TEST_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(HDE "${P}/minhook/src/hde/hde64.c")
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(HDE "${P}/minhook/src/hde/hde32.c")
else()
  message(FATAL_ERROR "Only x86 and x64 are supported")
endif()
add_library(process_hook STATIC "${P}/staging.c" "${P}/helper.c" "${P}/payload.c"
  "${P}/startup_hooks.c" "${P}/late_batch.c" "${P}/propagate.c" "${P}/unwind.c"
  "${P}/late_patch.c" "${P}/late_guard.c" "${P}/minhook/src/buffer.c"
  "${P}/minhook/src/trampoline.c" "${HDE}")
target_include_directories(process_hook PUBLIC "${ROOT}/src")
target_compile_definitions(process_hook PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_options(process_hook PRIVATE -fno-omit-frame-pointer -fno-optimize-sibling-calls -fno-lto)
endif()
enable_testing()
foreach(TEST unwind late_batch propagate)
  add_executable(${TEST} "${TEST_ROOT}/${TEST}.c")
  target_link_libraries(${TEST} PRIVATE process_hook)
  target_compile_definitions(${TEST} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${TEST} PRIVATE -fno-omit-frame-pointer -fno-optimize-sibling-calls -fno-lto)
    if("${TEST}" STREQUAL "propagate")
      target_link_options(${TEST} PRIVATE -municode)
    endif()
  endif()
  add_test(NAME ${TEST} COMMAND ${TEST})
  set_tests_properties(${TEST} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endforeach()
