# Actual native window tests: isolated, serialized, and separately labelled.
foreach(case kvm_window_modal kvm_window_retirement)
  set(name "nxvm-integration-${case}")
  add_executable(${name} "${ROOT}/test/integration/lib-nxvm/integration/${case}_smoke.c")
  target_link_libraries(${name} PRIVATE nxvm-test-providers)
  target_compile_options(${name} PRIVATE /UNDEBUG)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE LABELS "integration;desktop")
  list(APPEND nxvm_integration_tests ${name})
endforeach()
add_custom_target(shared-nxvm-integration DEPENDS ${nxvm_integration_tests})
