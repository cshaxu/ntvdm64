cmake_minimum_required(VERSION 3.24)
project(win31_editor NONE)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_program(NASM NAMES nasm.exe REQUIRED)
foreach(flag IN ITEMS WINDOWS95 GUEST_INPUT GUEST_ONLY FULLSCREEN BLOCKING DOS_FIRST FRONTEND FRONTEND_KEYS DISMISS CHILD_GEOMETRY)
  set(EDITOR_${flag} 0)
endforeach()
if(EDITOR_PROFILE STREQUAL "win95")
  set(EDITOR_WINDOWS95 1)
  set(MOUSE_MENU 0)
  set(DOSONLY 0)
  set(SAVE_WITNESS 0)
  set(FIRST_DOS 0)
  set(FIRST_DOS_BEFORE_SECOND 0)
  foreach(flag IN LISTS EDITOR_FEATURES)
    if(flag STREQUAL "MOUSE_MENU" OR flag STREQUAL "SAVE_WITNESS")
      set(${flag} 1)
    elseif(flag MATCHES "^(GUEST_INPUT|GUEST_ONLY|FULLSCREEN|BLOCKING|DOS_FIRST|FRONTEND|FRONTEND_KEYS|CHILD_GEOMETRY)$")
      set(EDITOR_${flag} 1)
    else()
      message(FATAL_ERROR "Unknown editor feature: ${flag}")
    endif()
  endforeach()
  if(EDITOR_GUEST_INPUT)
    set(MOUSE_MENU 1)
  endif()
  if(EDITOR_WELCOME_HEX)
    if(NOT EDITOR_WELCOME_HEX MATCHES "^([0-9a-fA-F][0-9a-fA-F])+$")
      message(FATAL_ERROR "Invalid welcome title hex")
    endif()
    string(LENGTH "${EDITOR_WELCOME_HEX}" size)
    if(size GREATER 126)
      message(FATAL_ERROR "Welcome title exceeds original limit")
    endif()
    string(REGEX MATCHALL ".." bytes "${EDITOR_WELCOME_HEX}")
    if("00" IN_LIST bytes)
      message(FATAL_ERROR "Welcome title contains zero")
    endif()
    list(TRANSFORM bytes PREPEND "0x")
    list(JOIN bytes "," EDITOR_WELCOME_BYTES)
    set(EDITOR_DISMISS 1)
  endif()
endif()
foreach(flag IN ITEMS MOUSE_MENU DOSONLY SAVE_WITNESS FIRST_DOS FIRST_DOS_BEFORE_SECOND)
  if(NOT "${${flag}}" MATCHES "^[01]$")
    message(FATAL_ERROR "Expected boolean ${flag}")
  endif()
endforeach()
if(DOSONLY AND (SAVE_WITNESS OR FIRST_DOS_BEFORE_SECOND))
  message(FATAL_ERROR "DOSONLY does not support save or first-editor sequence")
endif()
if(NOT IS_ABSOLUTE "${EDITOR_DEST}" OR EDITOR_DEST MATCHES "['\"\r\n]")
  message(FATAL_ERROR "Expected an absolute editor installation path")
endif()
set(editor "${repo}/test/integration/ntvdm64-386")
set(actions "${editor}/editor-actions.inc")
if(EDITOR_WINDOWS95)
  set(actions "${editor}/editor-win95-actions.inc")
endif()
set(extra)
foreach(name IN ITEMS production-key production-mouse welcome text-child blocking-actions win95-actions)
  list(APPEND extra "${editor}/editor-${name}.inc")
endforeach()
configure_file(config.inc.in config.inc @ONLY NEWLINE_STYLE LF)
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/controller.exe"
  COMMAND "${NASM}" -f bin -p "${CMAKE_BINARY_DIR}/config.inc"
    -o "${CMAKE_BINARY_DIR}/controller.exe" "${repo}/test/observation/win31_startup_contract.asm"
  DEPENDS "${CMAKE_BINARY_DIR}/config.inc" "${repo}/test/observation/win31_startup_contract.asm"
    "${editor}/editor-actions.inc" "${editor}/editor-routines.inc"
    "${editor}/editor-events.inc" "${editor}/editor-data.inc"
    "${editor}/frontend-mouse-request.inc" ${extra} "${NASM}" VERBATIM)
add_custom_target(controller ALL DEPENDS "${CMAKE_BINARY_DIR}/controller.exe")
