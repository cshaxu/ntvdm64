cmake_minimum_required(VERSION 3.24)
project(dos_identity NONE)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_program(NASM NAMES nasm.exe REQUIRED)
if(NOT DOS_IDENTITY_DEST MATCHES "^[A-Za-z]:[/\\]" OR DOS_IDENTITY_DEST MATCHES "['\";\r\n]")
  message(FATAL_ERROR "Expected an absolute DOS installation path")
endif()
set(startup "${repo}/test/observation/win31_startup_contract.asm")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${startup}")
file(READ "${startup}" source)
string(REPLACE "\r\n" "\n" source "${source}")
function(extract output start end)
  string(FIND "${source}" "${start}" first)
  if(first LESS 0)
    message(FATAL_ERROR "Missing identity source anchor: ${start}")
  endif()
  string(SUBSTRING "${source}" ${first} -1 rest)
  string(FIND "${rest}" "${end}" count)
  if(count LESS 0)
    message(FATAL_ERROR "Missing identity source boundary: ${end}")
  endif()
  string(SUBSTRING "${rest}" 0 ${count} value)
  set(${output} "${value}" PARENT_SCOPE)
endfunction()
string(FIND "${source}" "%ifdef W31_PROMPT_CHECK" count)
if(count LESS 0)
  message(FATAL_ERROR "Missing startup prefix boundary")
endif()
string(SUBSTRING "${source}" 0 ${count} prefix)
extract(helpers ".schedule_type:" "%endif\n.schedule_wait:")
extract(tail "push ds\npush word ready_name-data" "%ifdef W31_PROMPT_CHECK")
configure_file(controller.asm.in controller.asm @ONLY NEWLINE_STYLE LF)
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/controller.exe"
  COMMAND "${NASM}" -f bin "-DW31_INSTALL='${DOS_IDENTITY_DEST}'"
    "${CMAKE_BINARY_DIR}/controller.asm" -o "${CMAKE_BINARY_DIR}/controller.exe"
  DEPENDS "${CMAKE_BINARY_DIR}/controller.asm" "${NASM}" VERBATIM)
foreach(role IN ITEMS A B)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/${role}.COM"
    COMMAND "${NASM}" -f bin "-DW31_INSTALL='${DOS_IDENTITY_DEST}'" "-DROLE='${role}'"
      "${repo}/test/observation/dos_identity_child.asm" -o "${CMAKE_BINARY_DIR}/${role}.COM"
    DEPENDS "${repo}/test/observation/dos_identity_child.asm" "${NASM}" VERBATIM)
endforeach()
add_custom_target(identity ALL DEPENDS "${CMAKE_BINARY_DIR}/controller.exe"
  "${CMAKE_BINARY_DIR}/A.COM" "${CMAKE_BINARY_DIR}/B.COM")
