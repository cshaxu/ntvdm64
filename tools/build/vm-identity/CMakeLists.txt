cmake_minimum_required(VERSION 3.24)
project(vm_identity NONE)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_program(NASM NAMES nasm.exe REQUIRED)
if(IDENTITY_KIND STREQUAL "dos")
  set(source dos-identity.asm)
elseif(IDENTITY_KIND STREQUAL "retirement")
  set(source dos-retirement.asm)
elseif(IDENTITY_KIND STREQUAL "reuse")
  set(source dos-reuse.asm)
elseif(IDENTITY_KIND STREQUAL "controller")
  set(source win31-identity.asm)
else()
  message(FATAL_ERROR "Unknown identity probe")
endif()
foreach(flag IN LISTS IDENTITY_DEFINES)
  if(NOT flag MATCHES "^-D[A-Z_][A-Z0-9_]*=(1|437|850|sequence|repeat|function|extended|oem)$")
    message(FATAL_ERROR "Invalid identity define: ${flag}")
  endif()
endforeach()
set(source "${repo}/test/integration/ntvdm64-386/${source}")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/probe.bin"
  COMMAND "${NASM}" -f bin ${IDENTITY_DEFINES} "${source}" -o "${CMAKE_BINARY_DIR}/probe.bin"
  DEPENDS "${source}" "${NASM}" VERBATIM)
add_custom_target(probe ALL DEPENDS "${CMAKE_BINARY_DIR}/probe.bin")
