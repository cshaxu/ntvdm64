function(control_unit name source extra)
  add_library(control_${name} OBJECT EXCLUDE_FROM_ALL "${source}")
  set_property(TARGET control_${name} PROPERTY COMPILE_FLAGS "${control_flags} ${extra}")
endfunction()
function(control_link name extra)
  set(objects)
  set(dependencies)
  foreach(unit IN LISTS ARGN)
    list(APPEND objects "$<TARGET_OBJECTS:control_${unit}>")
    list(APPEND dependencies control_${unit})
  endforeach()
  separate_arguments(options WINDOWS_COMMAND "${extra}")
  set(byproducts)
  if(name MATCHES "[.]dll$")
    list(APPEND byproducts "${CMAKE_BINARY_DIR}/obj/destroy_event_probe.lib"
      "${CMAKE_BINARY_DIR}/obj/destroy_event_probe.exp")
  endif()
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/bin/${name}"
    COMMAND "${CMAKE_LINKER}" /nologo /machine:x86 "/out:bin/${name}" "/map:bin/${name}.map"
      ${objects} user32.lib gdi32.lib ${options}
    DEPENDS ${dependencies} ${objects} BYPRODUCTS "${CMAKE_BINARY_DIR}/bin/${name}.map" ${byproducts}
    COMMAND_EXPAND_LISTS VERBATIM)
endfunction()
