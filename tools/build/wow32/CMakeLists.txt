function(wow_unit name source flags)
  if(source MATCHES "[.](c|cxx)$")
    set_source_files_properties("${source}" PROPERTIES LANGUAGE C)
    string(APPEND flags " /O2")
  endif()
  add_library(${name} OBJECT "${source}")
  # Keep ordered, paired /I and /FI arguments intact; option de-duplication
  # would otherwise merge different include roots.
  set_property(TARGET ${name} PROPERTY COMPILE_FLAGS "${flags}")
endfunction()

foreach(raw IN LISTS source_names)
  string(STRIP "${raw}" source)
  set(flags "${wow_cflags}")
  if(source STREQUAL "wstruc.c")
    set(flags "${cflags}")
  elseif(source STREQUAL "wkfileio.c")
    set(flags "/I \"${NTVDM_REPO}/src/compat/include/wow-vdd\" ${flags}")
  elseif(source STREQUAL "wkman.c")
    string(APPEND flags " /FI \"${NTVDM_REPO}/src/compat/include/wow_native_process.h\"")
  elseif(source STREQUAL "wcommdlg.c")
    string(APPEND flags " /DWOW_COMMON_DIALOG_BINDING /FI \"${NTVDM_REPO}/src/compat/include/wow_common_dialog_calls.h\"")
  elseif(source STREQUAL "wkgthunk.c")
    string(APPEND flags " /DWOW_X86_GENERIC_CALL /DWK32ICallProc32=wow_original_WK32ICallProc32 /DWOWYield16=wow_original_WOWYield16 /DWOWDirectedYield16=wow_original_WOWDirectedYield16 /DWOW_NATIVE_EXTENSION_BINDING /FI \"${NTVDM_REPO}/src/compat/include/wow_user_native_extension.h\"")
  elseif(source STREQUAL "wmmedia.c")
    string(APPEND flags " /DWMM32CallProc32=wow_original_WMM32CallProc32")
  endif()
  if(source MATCHES "^(wow32|wudlg|wuser)[.]c$")
    string(APPEND flags " /DWOWSysErrorBox=wow_user_sys_error_box")
  endif()
  if(source STREQUAL "wshell.c")
    string(APPEND flags " /DWOWShellExecute=wow_shell_execute")
  endif()
  if(source MATCHES "^(wshell|wkernel|mapembed|wdde|wow32fax)[.]c$")
    string(APPEND flags " /FI \"${NTVDM_REPO}/src/compat/include/wow_registry_binding.h\"")
  endif()
  string(REPLACE ".c" "" name "${source}")
  wow_unit(original_${name} "${original}/${source}" "${flags}")
  list(APPEND objects "$<TARGET_OBJECTS:original_${name}>")
endforeach()
